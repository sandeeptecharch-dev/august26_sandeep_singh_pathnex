#Day 19, 20 August 2026 : Terrform create infra using terraform (provider, data [aws_subnets, aws_ami], resource [aws_security_group, aws_instance])


provider "aws" {
  region = "ap-south-1"
  # Uses AWS_PROFILE=pathnex from your terminal
}

# Use default VPC (simple)
data "aws_vpc" "default" {
  default = true
}

data "aws_subnets" "default" {
  filter {
    name   = "vpc-id"
    values = [data.aws_vpc.default.id]
  }
}

# Latest Amazon Linux 2023 AMI
data "aws_ami" "al2023" {
  most_recent = true
  owners      = ["amazon"]

  filter {
    name   = "name"
    values = ["al2023-ami-*-x86_64"]
  }
}

resource "aws_security_group" "web_sg" {
  name        = "pathnex-feb-tf-sg"
  description = "Allow HTTP only"
  vpc_id      = data.aws_vpc.default.id

  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  egress {
    description = "All outbound"
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Project = "terraform-class"
    Name    = "pathnex-feb-tf-sg"
  }
}

resource "aws_instance" "feb-tf" {
  ami                         = data.aws_ami.al2023.id
  instance_type               = var.instance_type
  subnet_id                   = data.aws_subnets.default.ids[0]
  vpc_security_group_ids      = [aws_security_group.web_sg.id]
  associate_public_ip_address = true

  user_data = <<-EOF
              #!/bin/bash
              dnf update -y
              dnf install -y httpd
              systemctl enable httpd
              systemctl start httpd
              echo "<h1>Hello All, Today is 18 Feb 2026 and ec2 creation with the help of terraform is successful </h1>" > /var/www/html/index.html
              EOF

  tags = {
    Project = "terraform-class-for-ec2"
    Name    = "pathnex-feb-tf-instance"
    batch   = var.name_prefix
  }
}
